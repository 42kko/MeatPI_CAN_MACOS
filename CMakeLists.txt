cmake_minimum_required(VERSION 3.16)
project(meatcan VERSION 0.1.0 LANGUAGES CXX)
if(NOT APPLE)
  message(FATAL_ERROR "meatcan supports macOS only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_path(LIBUSB_INCLUDE_DIR libusb.h PATH_SUFFIXES libusb-1.0 HINTS /opt/homebrew/opt/libusb/include /usr/local/opt/libusb/include REQUIRED)
find_library(LIBUSB_LIBRARY usb-1.0 HINTS /opt/homebrew/opt/libusb/lib /usr/local/opt/libusb/lib REQUIRED)
add_executable(meatcan src/main.cpp)
target_include_directories(meatcan PRIVATE include)
target_include_directories(meatcan SYSTEM PRIVATE ${LIBUSB_INCLUDE_DIR})
target_link_libraries(meatcan PRIVATE ${LIBUSB_LIBRARY})
target_compile_options(meatcan PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS meatcan RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol_test tests/protocol_test.cpp)
  target_include_directories(protocol_test PRIVATE include)
  add_test(NAME protocol COMMAND protocol_test)
endif()

if(BUILD_TESTING)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_integration.py $<TARGET_FILE:meatcan>)
    set_tests_properties(cli_integration PROPERTIES TIMEOUT 60)
  endif()
endif()
